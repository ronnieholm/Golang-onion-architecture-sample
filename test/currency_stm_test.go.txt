// Maintain lists of in-memory aggregates.
// After each step, replay events and compare with in-memory aggregates.

var ctx = context.Background()
var clock = &SwitchableClock{}
var dispatcher infrastructure.Dispatcher

var currencies = map[uuid.UUID]core.CurrencyResponse{}
var freeCurrencyCodes map[string]struct{}

func setup() {
	LoadConfig()
	dispatcher = infrastructure.NewDispatcher(ctx, *config, infrastructure.WithClock(clock))
	ResetDB(ctx, dispatcher.PgxPool)
	for cc := range core.CurrencyCodes {
		freeCurrencyCodes[cc] = struct{}{}
	}
}

func createCurrency(t *rapid.T) {
	if len(freeCurrencyCodes) == 0 {
		t.Skip("no free codes")
	}

	// clock.Current = clock. advance between 1 and 7 days.
	create := CreateCurrencyCommandGen().Draw(t, "create_currency")
	create.Code = genMapKey(freeCurrencyCodes, strings.Compare).Draw(t, "free_currency_code")
	delete(freeCurrencyCodes, create.Code)

	_, err := dispatcher.CreateCurrency(ctx, create)
	require.NoError(t, err)
	currency, err := dispatcher.GetCurrency(ctx, core.GetCurrencyQuery{Code: create.Code})
	require.NoError(t, err)
	currencies[currency.ID] = *currency
}
